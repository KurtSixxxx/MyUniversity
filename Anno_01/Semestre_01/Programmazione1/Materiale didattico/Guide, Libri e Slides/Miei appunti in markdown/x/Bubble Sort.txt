#include <stdio.h>
#include <stdbool.h>
#define N 5

int main(){
	int arr[N] = {8, 2, 3, 6, 9};
	for(int i = 0; i < N-1; i++){
		bool sentinella;
			for(int j = 0; j < N - i - 1; j++){
				if(arr[j] > arr[j+1]){
				int tmp = arr[j];
				arr[j] = arr[j+1];
				arr[j+1] = tmp;
				sentinella = 1;
				}
			}
		if(sentinella == 0) break;
		}
return 0;
}
