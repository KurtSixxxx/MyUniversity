PImage negativo(PImage I)
{
  PImage R=I.copy();
  
  float r,g,b;
  R.loadPixels();
  
  for(int i=0; i<R.pixels.length; i++)
  {
    r=255-red(R.pixels[i]);
    g=255-green(R.pixels[i]);
    b=255-blue(R.pixels[i]);
    
    R.pixels[i]=color(r,g,b);
  }
  return R;
}

PImage gamma(PImage I, float gm)
{
  PImage R=createImage(I.width,I.height, RGB);
  
  R.loadPixels();
  I.loadPixels();
  float r,g,b;
  float c=1/pow(255,gm-1);
  
  for(int i=0; i<R.pixels.length; i++)
  {
  
    r=c*pow(red(I.pixels[i]),gm);
    g=c*pow(green(I.pixels[i]),gm);
    b=c*pow(blue(I.pixels[i]),gm);
    
    R.pixels[i]=color(r,g,b);
  }
  
  R.updatePixels();
  return R;
}

PImage logaritmo(PImage I)
{
  PImage R=I.copy();
  
  R.loadPixels();
  I.loadPixels();
  
  float r,g,b;
  float C=255/log(256);
  
  for(int i=0; i<R.pixels.length; i++)
  {
    r=C*log(1+red(I.pixels[i]));
    g=C*log(1+green(I.pixels[i]));
    b=C*log(1+blue(I.pixels[i]));
    
    R.pixels[i]=color(r,g,b);
  }
  
  R.updatePixels();
  
  return R;
}

PImage stretching(PImage I)
{
  PImage R=I.copy();
  
  float max,min;
  R.loadPixels();
  
  max=red(R.pixels[0]);
  min=red(R.pixels[0]);
  
  for(int i=1; i<R.pixels.length; i++)
  {
    if(red(R.pixels[i])<min)
    min=red(R.pixels[i]);
    
    if(red(R.pixels[i])>max)
    max=red(R.pixels[i]);
  }
  
  for(int i=0; i<R.pixels.length; i++)
  {
    R.pixels[i]=color(255*(red(R.pixels[i])-min)/(max-min));
  }
  
  R.updatePixels();
  
  return R;
}


float [] istogramma(PImage I)
{
  float[] H=new float[256];
  for(int i=0; i<256; i++)
  {
    H[i]=0;
  }
  
  I.loadPixels();
  for(int i=0; i<I.pixels.length; i++)
  {
    H[int(red(I.pixels[i]))]++;
  }
  
  for(int i=0; i<256; i++)
  {
    H[i]=H[i]/I.pixels.length;
  }
  
  return  H;
  
}

  PImage R=I.copy();
  
  float [] H=istogramma(I);

{
  PImage R=I.copy();
  R.loadPixels();
  for(int i=0; i<R.pixels.length; i++)
  {
    R.pixels[i]=color(0);
  }
  
  float v,w;
  float x1,y1;
  
  for(int x=0; x<I.width; x++)
  {
    for(int y=0; y<I.height; y++)
    {
      x1=x-I.width/2;
      y1=y-I.height/2;
      
      v=x1*cos(theta)+y1*sin(theta);
      w=-x1*sin(theta)+y1*cos(theta);
      
      v=v+I.width/2;
      w=w+I.height/2;
      
      R.set(x,y,I.get(round(v),round(w)));
    
    }
  }
  
  R.updatePixels();
  
  return R;
}
